  value,

  onContinue,
  onSkip,
}: StepIdeaProps): React.JSX.Element {
  const trimmedLength = value.trim().length;
  const canProceed = trimmedLength >= MIN_CHARS;

    Eif (e.target.value.length <= MAX_CHARS) {

  return (
    <div className="flex flex-col">
